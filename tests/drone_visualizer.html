import asyncio, websockets, json

async def stream(env, host="localhost", port=8765):
  async with websockets.serve(
    lambda ws: sender(ws, env), host, port):
    await asyncio.Future()

async def sender(ws, env):
  obs, _ = env.reset()
  while True:
    action = policy(obs)
    obs, rew, done, _, _ = env.step(action)
    frame = build_frame(env)  # see docs
    await ws.send(json.dumps(frame))
    if done: break
